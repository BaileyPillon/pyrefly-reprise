def lens_table(L, names):
    out = {}
    for shot_name, figs in L.items():
        if shot_name not in names:
            continue
        out[names[shot_name]] = {
            fid: {'degOffLens': v['off'], 'widthKept': v['widthKept'], 'painting': v['painting'], 'shareInFrame': v['inFrame']}
            for fid, v in figs.items()
        }
    return out
