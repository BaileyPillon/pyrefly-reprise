W.tick = () => {
  try {
    if (!W.s0 || W.b() !== W.s0.b) return;
    const cam = W.camera().position;
    for (const [id, o] of Object.entries(W.over)) {
      const a = W.act(id); if (!a) continue; const u = a.u;
      if (o.face) a.rotation.y = Math.atan2(cam.x - a.position.x, cam.z - a.position.z) + (o.yawOff || 0);
      else if (o.yaw != null) a.rotation.y = o.yaw;
      if (o.tint != null) u.tint.value.setHex(o.tint);
      if (o.brightness != null) u.brightness.value = o.brightness;
      if (o.desaturate != null) u.desaturate.value = o.desaturate;
      if (o.rim != null) u.rimColor.value.setHex(o.rim);
      if (o.rimStrength != null) u.rimStrength.value = o.rimStrength;
      if (o.rimDir) u.rimDir.value.set(o.rimDir[0] * (a.mirrored ? -1 : 1), o.rimDir[1]);
      if (o.rimWidth != null) u.rimWidth.value = o.rimWidth;
      if (o.bounce != null) u.bounceColor.value.setHex(o.bounce);
      if (o.bounceStrength != null) u.bounceStrength.value = o.bounceStrength;
    }
    for (const f of W.ticks || []) f();
  } catch (e) { W.err = String(e); }
};

// An out-of-focus foreground silhouette (rock, ice crystal, pillar) as a camera-facing card with a pre-blurred texture.

  const grd = g.createLinearGradient(0, 0, 0, c.height); grd.addColorStop(0, o.top ?? o.color ?? '#0b1424'); grd.addColorStop(1, o.color ?? '#0b1424');

  if (o.glint) { g.filter = `blur(${(o.blur ?? 14) * 0.6}px)`; g.strokeStyle = o.glint; g.lineWidth = 6; g.beginPath(); o.pts.slice(0, o.glintN ?? 3).forEach(([px, py], i) => { const X = pad + px * (S - 2 * pad), Y = pad + py * (c.height - 2 * pad); if (i) g.lineTo(X, Y); else g.moveTo(X, Y); }); g.stroke(); }

// A camera-facing painted card placed in screen terms: centre at NDC (x, y), d from the lens, hFrac of the frame height tall.
W.faceCard = async (o) => {
  const T = await W.T(); const tex = await new T.TextureLoader().loadAsync(o.url); tex.colorSpace = T.SRGBColorSpace; tex.anisotropy = 8;
  const cam = W.camera(); const d = o.d ?? 1.5; const fh = 2 * d * Math.tan((cam.fov * Math.PI) / 360); const h = fh * (o.hFrac ?? 1.1); const w = h * tex.image.width / tex.image.height;
  const m = new T.Mesh(new T.PlaneGeometry(w, h), new T.MeshBasicMaterial({ map: tex, transparent: true, depthWrite: false, depthTest: false, fog: false, toneMapped: false }));
  if (o.mirror) m.scale.x = -1;
  m.position.fromArray(await W.ndc(o.at[0], o.at[1], d)); m.renderOrder = 85; m.name = 'persp-face'; m.onBeforeRender = (r, s, c) => m.quaternion.copy(c.quaternion);
  W.scene().add(m); W.added.push(m); return [w, h];
};

// A theatre spotlight: a soft additive cone (narrow at the top, wide at the floor) standing over a point, facing the camera.
W.spotCone = async (o) => {
  const T = await W.T(); const c = document.createElement('canvas'); c.width = 256; c.height = 512; const g = c.getContext('2d');
  g.filter = 'blur(10px)'; const gr = g.createLinearGradient(0, 0, 0, 512); gr.addColorStop(0, 'rgba(255,255,255,0.05)'); gr.addColorStop(0.7, 'rgba(255,255,255,0.45)'); gr.addColorStop(1, 'rgba(255,255,255,0.8)');
  g.fillStyle = gr; g.beginPath(); g.moveTo(128 - 18, 0); g.lineTo(128 + 18, 0); g.lineTo(256 - 20, 512); g.lineTo(20, 512); g.closePath(); g.fill();
  const tex = new T.CanvasTexture(c); tex.colorSpace = T.SRGBColorSpace;
  const m = new T.Mesh(new T.PlaneGeometry(o.w ?? 2.4, o.h ?? 7), new T.MeshBasicMaterial({ map: tex, color: o.color ?? 0xffe2a8, transparent: true, opacity: o.o ?? 0.3, depthWrite: false, blending: T.AdditiveBlending, fog: false, toneMapped: false }));
  m.position.set(o.p[0], (o.h ?? 7) / 2 + (o.y0 ?? 0), o.p[2]); m.renderOrder = 9; m.name = 'persp-cone';
  m.onBeforeRender = (r, s, cam) => { m.rotation.set(0, Math.atan2(cam.position.x - m.position.x, cam.position.z - m.position.z), 0); };
  W.scene().add(m); W.added.push(m); return m;
};
