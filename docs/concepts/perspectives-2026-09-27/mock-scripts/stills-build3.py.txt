def get(key, opt): return next(f for f in FJ['frames'] if f['chapter'] == key and f['option'] == opt)
def chip(d, x, y, tag, sz=13):
    tf = font(BOLD, sz); tw = d.textlength(tag, font=tf); h = sz + 6
    d.rectangle([x, y, x + tw + 12, y + h], fill=tagcol(tag)); d.text((x + 6, y + 1), tag, font=tf, fill=PAPER); return tw + 12
def label(d, x0, y0, w, f, h=58):
    d.rectangle([x0, y0, x0 + w, y0 + h], fill=INK); d.text((x0 + 10, y0 + 3), f"{NUM[f['option']]} · {f['name']}", font=font(BOLD, 21), fill=PAPER)
    tw = chip(d, x0 + 10, y0 + 33, f['provenance'])
    if f['standIns']:
        s = 'STAND-IN: ' + f['standIns'][0]; sf = font(REG, 13); maxw = w - (tw + 32)
        if d.textlength(s, font=sf) > maxw:
            words = s.split(' ')
            while words and d.textlength(' '.join(words) + ' …', font=sf) > maxw: words.pop()
            s = ' '.join(words).rstrip(',;:') + ' …'
        d.text((x0 + 18 + tw, y0 + 35), s, font=sf, fill=PINK)
def header(S, d, title, lines, h):
    d.rectangle([0, h - 4, S.width, h], fill=GOLD); d.text((24, 14), title, font=font(BOLD, 30), fill=PAPER); y = 58
    for ln in lines: d.text((24, y), ln, font=font(REG, 18), fill=GREY); y += 24
def save_sheet(S, name, limit=2_600_000):
    path = f'{OUT}/sheets/{name}'; q = 86
    while True:
        S.save(path, quality=q, optimize=True, progressive=True)
        if os.path.getsize(path) < limit or q < 60: break
        q -= 6
    print(name, S.size, os.path.getsize(path) // 1024, 'KB q', q)

def stage2_sheet(key, name):
    order = ['P2', 'P2h', 'P2l', 'P2wd', 'P2r', 'P3', 'P21', 'P22', 'P10', 'P4', 'P13', 'P6', 'P9']
    fs = [f for f in FJ['frames'] if f['chapter'] == key and f['option'] in order]
    fs.sort(key=lambda f: order.index(f['option']))
    cw, chh, lab, cols, head = 640, 360, 58, 4, 118
    rows = (len(fs) + cols - 1) // cols; bottom = 640
    S = Image.new('RGB', (cw * cols, head + rows * (chh + lab) + bottom), INK); d = ImageDraw.Draw(S)
    header(S, d, f'Battle perspectives · stage 2 · {CH[key]["title"]}', ['Real-engine frames at 1920x1080 (phone 390x844 at DPR 3) from the worktree, no src/ change. Rear paintings, far-side plates, boss rear views, the arch and the',
                                                                          "floor plates are the art agent's CANDIDATES (art/READY.json). P4, P13, P6" + (' and P9' if key == 'ffx-ch1' else '') + " now use the painted art; their stage-1 stand-ins are frames/alt/*-standin.jpg."], head)
    for i, f in enumerate(fs):
        x0 = (i % cols) * cw; y0 = head + (i // cols) * (chh + lab)
        S.paste(Image.open(f"{OUT}/{f['file']}").convert('RGB').resize((cw, chh), Image.LANCZOS), (x0, y0)); label(d, x0, y0 + chh, cw, f)
    yb = head + rows * (chh + lab)
    ph_ = get(key, 'P15'); pim = Image.open(f"{OUT}/{ph_['file']}").convert('RGB'); pw = round(pim.width * (bottom - 70) / pim.height)
    S.paste(pim.resize((pw, bottom - 70), Image.LANCZOS), (20, yb + 6)); label(d, 0, yb + bottom - 62, pw + 40, ph_)
    st = get(key, 'P16'); sim = Image.open(f"{OUT}/{st['file']}").convert('RGB'); sw = S.width - pw - 80; sh = round(sim.height * sw / sim.width)
    S.paste(sim.resize((sw, sh), Image.LANCZOS), (pw + 60, yb + 60)); label(d, pw + 60, yb + 60 + sh + 8, 900, st)
    d.text((pw + 60, yb + 16), "P16 · Director's Cut: a camera language across five beats (HUD down after the command beat)", font=font(BOLD, 22), fill=GOLD)
    save_sheet(S, name)

def final_sheet():
    fam = lambda key: [('Today', ['P0', 'P0m', 'P1']), ('Over the shoulder', ['P2-without', 'P2', 'P2h', 'P2l', 'P2wd', 'P2r', 'P3', 'P21']),
                       ('Other cameras', ['P4', 'P5', 'P5m', 'P6', 'P7', 'P8', 'P9' if key == 'ffx-ch1' else 'P10']),
                       ('Frames and moments', ['P11', 'P12', 'P13', 'P14', 'P22']), ('Phone and sequence', ['P15', 'P16'])]
    cw, chh, lab, gap, lm, cols = 336, 189, 42, 6, 180, 8
    rowh = chh + lab + gap; chtitle = 52; head = 124
    W_ = lm + cols * cw; H_ = head + 2 * (chtitle + 5 * rowh) + 10
    S = Image.new('RGB', (W_, H_), INK); d = ImageDraw.Draw(S)
    header(S, d, 'Battle perspectives · every kept option, both chapters (final)', ['Real-engine frames from the worktree (no src/ change), the lead\'s first command menu, seed 1. Provenance chip under each frame: green = existing art, blue = new',
                                                                                     "candidate art from art/READY.json, red = composite. Details, cameras, measurements and stand-ins: frames/frames.json. P20 (canon reconstruction) is on hold."], head)
    y = head
    for key in CH:
        d.text((24, y + 10), CH[key]['title'], font=font(BOLD, 28), fill=GOLD); y += chtitle
        for fname, opts in fam(key):
            words, lines, ff = fname.split(' '), [''], font(BOLD, 20)
            for w_ in words:
                t = (lines[-1] + ' ' + w_).strip()
                if d.textlength(t, font=ff) <= lm - 34: lines[-1] = t
                else: lines.append(w_)
            for i_, ln in enumerate(lines): d.text((18, y + 12 + i_ * 26), ln, font=ff, fill=PAPER)
            d.rectangle([18, y + 18 + len(lines) * 26, 150, y + 20 + len(lines) * 26], fill=GOLD)
            col = 0
            for o in opts:
                f = get(key, o); x0 = lm + col * cw
                if o == 'P15':
                    cell = Image.new('RGB', (cw - 4, chh), (8, 6, 12)); pim = Image.open(f"{OUT}/{f['file']}").convert('RGB'); pw = round(pim.width * chh / pim.height)
                    cell.paste(pim.resize((pw, chh), Image.LANCZOS), ((cw - 4 - pw) // 2, 0)); S.paste(cell, (x0, y)); span = 1
                elif o == 'P16':
                    span = 5; sim = Image.open(f"{OUT}/{f['file']}").convert('RGB'); sw = round(sim.width * chh / sim.height)
                    if sw > span * cw - 4: sw = span * cw - 4
                    sh = round(sim.height * sw / sim.width); S.paste(sim.resize((sw, sh), Image.LANCZOS), (x0, y + (chh - sh) // 2))
                else:
                    S.paste(Image.open(f"{OUT}/{f['file']}").convert('RGB').resize((cw - 4, chh), Image.LANCZOS), (x0, y)); span = 1
                d.text((x0 + 4, y + chh + 2), f"{NUM[o]} · {f['name']}", font=font(BOLD, 15), fill=PAPER)
                chip(d, x0 + 4, y + chh + 22, f['provenance'], 11)
                col += span
            y += rowh
    save_sheet(S, '06-final-all.jpg', 3_000_000)

stage2_sheet('ffx-ch1', '03-stage2-ffx.jpg')
