// ===== search-attack3.js =====
const M = __motion; const st = M.st(); const W = innerWidth, H = innerHeight;
const T = M.act('tidus').position, MO = M.act('mortiorchis').position;
const rect = (id) => { const r = st.projectRect(id); return r ? { x0: r.x / W, x1: (r.x + r.w) / W, y0: r.y / H, y1: (r.y + r.h) / H } : null; };
const res = [];
for (let cx = -3.0; cx <= 0.2; cx += 0.2) for (let cz = 4.4; cz <= 8.0; cz += 0.4) for (const cy of [1.0, 1.2, 1.4, 1.6, 1.8, 2.0]) for (const w of [0.25, 0.4, 0.55]) for (const ly of [1.2, 1.45, 1.7]) {
  const look = [T.x + (MO.x - T.x) * w, ly, T.z + (MO.z - T.z) * w];
  M.set('cam', { pos: [cx, cy, cz], look, fov: 32, roll: 0, sway: 0 }); M.apply(0);
  const vis = st.visibility();
  const t = rect('tidus'), mo = rect('mortiorchis');
  if (!t || !mo) continue;
  const mv = vis.get('mortiorchis') ?? 1, tv = vis.get('tidus') ?? 1;
  const th = t.y1 - t.y0, tc = (t.x0 + t.x1) / 2, mc = (mo.x0 + mo.x1) / 2;
  const az = Math.abs(Math.atan2(cx - T.x, cz - T.z) * 57.3);
  let s = 0;
  if (mv < 0.8) s -= 5 + (0.8 - mv) * 10;
  if (tv < 0.95) s -= 3;
  s -= Math.abs(th - 0.66) * 4;                          // attacker big
  s -= Math.max(0, 0.40 - t.x0) * 5;                     // clear of the command list
  s -= Math.max(0, tc - 0.58) * 4;
  s -= Math.max(0, mc - 0.8) * 3;
  s -= (cy - 1.0) * 0.6;                                 // lower is better (the shot is 'low')
  s -= Math.max(0, az - 25) * 0.1;
  s -= Math.max(0, 0.08 - t.y0) * 3;                     // head not cut
  res.push({ s: +s.toFixed(3), pos: [+cx.toFixed(2), cy, +cz.toFixed(2)], look: look.map((v) => +v.toFixed(2)), mv: +mv.toFixed(2), th: +th.toFixed(2), t: [t.x0, t.y0, t.x1, t.y1].map(v => +v.toFixed(2)), mo: [mo.x0, mo.y0, mo.x1, mo.y1].map(v => +v.toFixed(2)), az: +az.toFixed(0) });
}
res.sort((a, b) => b.s - a.s); M.cut('master'); M.apply(0);
return res.slice(0, 10);

// ===== search-rev.js =====
const M = __motion; const st = M.st(); const W = innerWidth, H = innerHeight;
const rect = (id) => { const r = st.projectRect(id); return r ? { x0: r.x / W, x1: (r.x + r.w) / W, y0: r.y / H, y1: (r.y + r.h) / H } : null; };
M.cut('rev');
const party = ['tidus', 'kimahri'];
const res = [];
for (let cx = 4; cx <= 13; cx += 0.75) for (let cz = -19; cz <= -8.5; cz += 0.75) for (const cy of [1.5, 2.0]) for (const fov of [20, 24, 28]) for (const lx of [-1.2, -0.5, 0.2]) {
  const look = [lx, 1.15, 0.4];
  M.set('cam', { pos: [cx, cy, cz], look, fov, roll: 0, sway: 0 }); M.apply(0); M.faceTick();
  for (const id of party) M.act(id).updateMatrixWorld(true);
  const vis = st.visibility(); const se = rect('seymour-flux'), mo = rect('mortiorchis');
  const rs = party.map(rect); if (rs.some((r) => !r)) continue;
  const px0 = Math.min(...rs.map((r) => r.x0)), px1 = Math.max(...rs.map((r) => r.x1)), ph = Math.max(...rs.map((r) => r.y1 - r.y0));
  const pv = Math.min(...party.map((id) => vis.get(id) ?? 1));
  const pc = (px0 + px1) / 2;
  let s = 0;
  s -= Math.max(0, 0.9 - pv) * 12;
  s -= Math.abs(pc - 0.45) * 3;
  s -= Math.abs(ph - 0.38) * 4;
  const seIn = se ? Math.max(0, Math.min(1, se.x1) - Math.max(0, se.x0)) : 0;
  s -= se ? (Math.abs(Math.max(se.x0, 0) - 0.72) * 2) : 3;       // Seymour's near edge around 72% of the width
  s -= seIn < 0.12 ? 2 : 0;                                        // and actually in frame
  s -= px1 > 0.8 ? (px1 - 0.8) * 5 : 0; s -= px0 < 0.2 ? (0.2 - px0) * 5 : 0;
  res.push({ s: +s.toFixed(3), pos: [cx, cy, cz], look, fov, pv: +pv.toFixed(2), pc: +pc.toFixed(2), ph: +ph.toFixed(2), se: se && [se.x0, se.y0, se.x1, se.y1].map(v => +v.toFixed(2)), mo: mo && [mo.x0, mo.x1].map(v => +v.toFixed(2)) });
}
res.sort((a, b) => b.s - a.s); M.cut('master'); M.apply(0);
return res.slice(0, 8);

// ===== fit-side.js =====
const M = __motion; const st = M.st(); const W = innerWidth;
M.cut('side');
const T = M.act('tidus').position, S = M.act('seymour-flux').position;
const out = [];
for (const cam of [[18.2, 0.9, 7.6], [16.5, 0.95, 6.5], [15.0, 1.0, 5.8], [20.0, 1.0, 9.0]]) for (const fov of [20, 22, 24]) {
  let best = null;
  for (let w = 0; w <= 1.0001; w += 0.05) {
    const look = [T.x + (S.x - T.x) * w, 1.55, T.z + (S.z - T.z) * w];
    M.set('cam', { pos: cam, look, fov, roll: 0, sway: 0 }); M.apply(0); M.faceTick();
    for (const id of st.staged()) M.act(id).updateMatrixWorld(true);
    const t = st.projectRect('tidus'), s = st.projectRect('seymour-flux'), k = st.projectRect('kimahri'), y = st.projectRect('yuna');
    if (!t || !s) continue;
    const px0 = Math.min(t.x, k.x, y.x) / W, sx1 = (s.x + s.w) / W;
    const mid = (px0 + sx1) / 2;
    const e = Math.abs(mid - 0.45) + Math.max(0, 0.18 - px0) + Math.max(0, sx1 - 0.78);
    if (!best || e < best.e) best = { e: +e.toFixed(3), cam, fov, look: look.map(v => +v.toFixed(2)), px0: +px0.toFixed(2), sx1: +sx1.toFixed(2), th: +(t.h / innerHeight).toFixed(2), sh: +(s.h / innerHeight).toFixed(2) };
  }
  out.push(best);
}
M.cut('master'); M.apply(0);
return out.sort((a, b) => a.e - b.e);

// ===== search-phone3.js =====

const M = __motion; const st = M.st();
const cv = document.querySelector('canvas[data-role=game-canvas]').getBoundingClientRect();
const vis0 = (0 - cv.x) / cv.width, vis1 = (innerWidth - cv.x) / cv.width;           // visible band, render-normalised

const R = (id) => { const r = st.projectRect(id); if (!r) return null; return { x0: (r.x - cv.x) / cv.width, x1: (r.x + r.w - cv.x) / cv.width, y0: (r.y - cv.y) / cv.height, y1: (r.y + r.h - cv.y) / cv.height }; };
const T = M.act('tidus').position, S = M.act('seymour-flux').position;
const band = (u) => (u - vis0) / (vis1 - vis0);                                          // 0..1 across the phone
const out = { vis: [vis0, vis1], cardsTop: cardsTop / cv.height };
const res = [];
for (let cx = -4.0; cx <= 1.0; cx += 0.25) for (let cz = 3.2; cz <= 8.0; cz += 0.4) for (const cy of [1.0, 1.25, 1.5]) for (const w of [0.1, 0.25, 0.4]) {

  M.set('cam', { pos: [cx, cy, cz], look, fov: 32, roll: 0, sway: 0 }); M.apply(0);
  const t = R('tidus'), s = R('seymour-flux'); if (!t || !s) continue;
  const vis = st.visibility().get('seymour-flux') ?? 1;
  const tc = band((t.x0 + t.x1) / 2), sc = band((s.x0 + s.x1) / 2), th = t.y1 - t.y0;
  let sc0 = 0;
  sc0 -= Math.abs(tc - 0.38) * 4 + Math.max(0, 0.62 - sc) * 3 + Math.max(0, sc - 0.98) * 3;

  sc0 -= Math.max(0, 0.02 - t.y0) * 4;
  sc0 -= Math.max(0, 0.5 - vis) * 8;
  sc0 -= (cy - 1.0) * 0.4;
  res.push({ s: +sc0.toFixed(3), pos: [+cx.toFixed(2), cy, +cz.toFixed(2)], look: look.map((v) => +v.toFixed(2)), tc: +tc.toFixed(2), sc: +sc.toFixed(2), th: +th.toFixed(2), vis: +vis.toFixed(2), feet: +t.y1.toFixed(2) });
}
res.sort((a, b) => b.s - a.s); out.atk = res.slice(0, 5);
// Items: push in on Tidus, centred on the phone
const res2 = [];
for (let cx = -2.2; cx <= 0.2; cx += 0.2) for (let cz = 2.6; cz <= 5.0; cz += 0.2) for (const cy of [1.45, 1.6]) {
