def assemble(clip):
    import shutil
    dst = f'{FRAMES}/_{clip}'
    if os.path.isdir(dst): shutil.rmtree(dst)
    os.makedirs(dst)
    n = 0; marks = []
    for src, a, b in EDITS[clip]:
        marks.append((n, src, a, b))
        for f in range(a, b + 1):
            shutil.copyfile(f'{FRAMES}/{src}/{f:05d}.jpg', f'{dst}/{n:05d}.jpg'); n += 1
    json.dump({'segments': [{'at': m[0], 'src': m[1], 'from': m[2], 'to': m[3]} for m in marks], 'frames': n}, open(f'{dst}/edit.json', 'w'), indent=1)
    print(clip, 'assembled', n, 'frames', [m[0] for m in marks])

def encode(clip, fps=30, limit=7.6e6):
    base = f'{FRAMES}/_{clip}' if clip in EDITS else f'{FRAMES}/{clip}'
    src = f'{base}/%05d.jpg'
    dst = f'{OUT}/{clip}.mp4'
    for crf in (21, 23, 25, 27, 29):
        # JPEG frames are full-range BT.601 YCbCr; ship standard limited-range BT.709 4:2:0 (yuvj420p plays badly in some browsers).
        vf = 'format=rgb24,scale=out_color_matrix=bt709:out_range=tv,format=yuv420p'
        subprocess.run([FFMPEG, '-v', 'error', '-y', '-framerate', str(fps), '-i', src, '-vf', vf, '-c:v', 'libx264', '-preset', 'slow',
                        '-tune', 'film', '-crf', str(crf), '-color_range', 'tv', '-colorspace', 'bt709', '-color_primaries', 'bt709',
                        '-color_trc', 'bt709', '-movflags', '+faststart', dst], check=True)
        size = os.path.getsize(dst)
        if size <= limit: break
    n = len([f for f in os.listdir(base) if f.endswith('.jpg')])
    print(clip, 'mp4', round(size / 1e6, 2), 'MB crf', crf, 'frames', n, 'dur', round(n / fps, 2))
    return {'crf': crf, 'bytes': size, 'frames': n, 'seconds': round(n / fps, 2)}

def strip(clip, fps=30):
    title, sub, keys = STRIPS[clip]
    cols = 3
    fw, fh = 640, 360
    if clip.endswith('-phone'):
        cols = len(keys); fw, fh = 300, 649
    rows = (len(keys) + cols - 1) // cols
    head, cap = 96, 40
    W = cols * fw + (cols + 1) * 12
    H = head + rows * (fh + cap + 12) + 12
    S = Image.new('RGB', (W, H), INK)
    d = ImageDraw.Draw(S)
    d.text((16, 12), title, font=font(BOLD, 30), fill=PAPER)
    d.text((16, 54), sub, font=font(REG, 19), fill=GREY)
    d.rectangle([0, head - 4, W, head - 2], fill=GOLD)
    base = f'{FRAMES}/_{clip}' if clip in EDITS else f'{FRAMES}/{clip}'
    nfr = len([f for f in os.listdir(base) if f.endswith('.jpg')])
    for i, (t, label) in enumerate(keys):
        n = min(int(round(t * fps)), nfr - 1)
        im = Image.open(f'{base}/{n:05d}.jpg').convert('RGB').resize((fw, fh), Image.LANCZOS)
        x = 12 + (i % cols) * (fw + 12)
        y = head + 8 + (i // cols) * (fh + cap + 12)
        S.paste(im, (x, y))
        d.rectangle([x, y + fh, x + fw, y + fh + 3], fill=GOLD)
        d.text((x + 2, y + fh + 7), f'{label}', font=font(BOLD, 18), fill=PAPER)
        d.text((x + fw - 62, y + fh + 9), f't={t:.1f}s', font=font(REG, 15), fill=GREY)
    p = f'{OUT}/{clip}-strip.jpg'
    S.save(p, quality=88, optimize=True, progressive=True)
    print(clip, 'strip', S.size, os.path.getsize(p) // 1024, 'KB')

        if c in STRIPS: strip(c)
