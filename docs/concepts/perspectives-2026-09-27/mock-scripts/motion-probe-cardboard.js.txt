// For each shot: how far each painted plane is turned away from the lens (0 = square on, 90 = edge on,
// >90 = the camera sees the plane's back, i.e. the painting mirrored). Raw = the engine's own eased yaw only;
// "turned" = with the shot's enter() hook (faceLens) applied, as filmed.
const M = __motion; const st = M.st();
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
const read = () => { const c = M.cam(); const o = {}; for (const id of st.staged()) { const a = M.act(id); if (!a || !a.visible) continue;
  const az = Math.atan2(c.position.x - a.position.x, c.position.z - a.position.z) * 180 / Math.PI;
  const yaw = (a.rotation.y + (a.inner?.rotation.y ?? 0)) * 180 / Math.PI; let d = az - yaw; while (d > 180) d -= 360; while (d < -180) d += 360;
  o[id] = { off: Math.round(d), eased: +(a.yawDeg ?? 0).toFixed(1), width: +Math.max(0, Math.cos(d * Math.PI / 180)).toFixed(2), back: Math.abs(d) > 90 }; } return o; };
const out = {};
for (const name of JSON.parse(SHOTS)) {
  const s = M.shots[name]; if (!s) continue;
  M.reset(); M.direct({ ...s, enter: undefined }); M.cut({ ...s, enter: undefined }); await sleep(1300);
  const raw = read();
  let turned = null;
  if (s.enter) { M.cut(s); await sleep(300); turned = read(); }
  out[name] = { raw, turned };
}
M.reset(); M.direct('master'); M.release(); M.reset();
return out;
