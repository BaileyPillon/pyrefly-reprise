// __motion: the MOTION rig's staging library for the battle-perspectives options round (2026-09-27).
// Console-only. It is evaluated into the running game page by the REPL (/eval wraps it in an async
// function) and is never imported by src/. Everything it does is a real-time technique the engine could
// ship: camera keyframes (position / look-at / fov / roll, cubic-bezier eases, an optional crane arc) on
// top of the game's own idle sway; an elliptical focus-blur pass (a stand-in for depth of field) and a
// directional whip-pan blur pass inserted into the live EffectComposer before the grade; the scene's own
// tilt-shift band; actor plane turns, silhouettes and painting swaps; a mirrored stand-in plate behind the
// field for reverse angles; per-frame re-projection of the target reticle; HUD show/hide.
const P = window.__pyrefly;
const M = window.__motion || (window.__motion = {});
M.b = () => P.battle();
M.st = () => P.battle().stage;
M.bc = () => P.battle().stage.opts.battleCamera;

// ------------------------------------------------------------------ easing (CSS cubic-bezier semantics)
M.bez = (x1, y1, x2, y2) => {
  const cx = 3 * x1, bx = 3 * (x2 - x1) - cx, ax = 1 - cx - bx;
  const cy = 3 * y1, by = 3 * (y2 - y1) - cy, ay = 1 - cy - by;
  const sx = (t) => ((ax * t + bx) * t + cx) * t;
  const sy = (t) => ((ay * t + by) * t + cy) * t;
  return (x) => {
    if (x <= 0) return 0;
    if (x >= 1) return 1;
    let lo = 0, hi = 1, t = x;
    for (let i = 0; i < 40; i++) { const v = sx(t); if (Math.abs(v - x) < 1e-6) break; if (v < x) lo = t; else hi = t; t = (lo + hi) / 2; }
    return sy(t);
  };
};
M.E = {
  move: M.bez(0.5, 0.0, 0.12, 1.0),   // menu camera: brisk start, long soft landing
  back: M.bez(0.45, 0.0, 0.2, 1.0),   // return to master
  inout: M.bez(0.65, 0.0, 0.35, 1.0),
  out: M.bez(0.16, 1.0, 0.3, 1.0),
  whip: M.bez(0.85, 0.0, 0.15, 1.0),  // whip pan: slow-fast-slow, most travel in the middle third
  drift: M.bez(0.37, 0.0, 0.63, 1.0),
  lin: (t) => t,
};
const ease = (e) => (typeof e === 'function' ? e : M.E[e] || M.E.move);

// ------------------------------------------------------------------ generic channels
M.t = M.t || 0;
M.ch = M.ch || {};
const clone = (v) => (v == null ? v : JSON.parse(JSON.stringify(v)));
const lerpAny = (a, b, p) => {
  if (typeof a === 'number' && typeof b === 'number') return a + (b - a) * p;
  if (Array.isArray(a) && Array.isArray(b)) return a.map((v, i) => lerpAny(v, b[i], p));
  if (a && b && typeof a === 'object' && typeof b === 'object') { const o = {}; for (const k of Object.keys(b)) o[k] = k in a ? lerpAny(a[k], b[k], p) : b[k]; return o; }
  return p < 0.5 ? a : b;
};
M.val = (name) => {
  const c = M.ch[name];
  if (!c) return undefined;
  if (!c.dur) return c.to;
  const k = Math.min(1, Math.max(0, (M.t - c.t0) / c.dur));
  const p = ease(c.ease)(k);
  const v = lerpAny(c.from, c.to, p);
  if (c.arc && name === 'cam') { const a = 4 * p * (1 - p); v.pos = v.pos.map((x, i) => x + c.arc[i] * a); }
  return v;
};
M.moving = (name) => { const c = M.ch[name]; return !!(c && c.dur && M.t - c.t0 < c.dur + 0.05); };
M.set = (name, v) => { M.ch[name] = { from: clone(v), to: clone(v), t0: M.t, dur: 0 }; };
M.tween = (name, to, ms, e, arc) => {
  const from = name === 'focus' ? (M.focusNow() ?? { ...M.FOCUS_OFF }) : (M.val(name) ?? clone(to));
  M.ch[name] = { from: clone(from), to: clone(to), t0: M.t, dur: Math.max(0.001, ms / 1000), ease: e, arc };
};

// ------------------------------------------------------------------ the camera
M.sway = { amp: 0.055, speed: 0.42, clock: 0, scale: 1 };
M.rigShot = (name) => { const r = M.bc().getRig(name); return { pos: [r.position.x, r.position.y, r.position.z], look: [r.lookAt.x, r.lookAt.y, r.lookAt.z], fov: r.fov ?? M.cam().fov, roll: 0, sway: r.sway ?? 1 }; };
M.now = () => { const c = M.cam(); const d = new c.position.constructor(); c.getWorldDirection(d); return { pos: c.position.toArray(), look: c.position.clone().add(d.multiplyScalar(10)).toArray(), fov: c.fov, roll: 0, sway: 1 }; };
M.shots = M.shots || {};
const shotOf = (s) => (typeof s === 'string' ? M.shots[s] : s);
/** Take over the camera (director mode) from `start` (a shot or shot name); the game's rigs stop driving it. */
M.direct = (start = 'master') => { M.hook(); M.on = true; M.set('cam', shotOf(start)); M.apply(0); return true; };
M.release = () => { M.on = false; };
/** Move to a shot. opts: {ms, ease, arc:[dx,dy,dz], focus, tilt, grade}. */
M.go = (s, opts = {}) => {
  const shot = shotOf(s);
  if (shot.enter) { try { shot.enter(); } catch (e) { M.errs.push('enter ' + e); } }
  const ms = opts.ms ?? 450;
  M.tween('cam', shot, ms, opts.ease ?? 'move', opts.arc ?? shot.arc);
  if (shot.focus || opts.focus) M.tween('focus', { ...M.FOCUS_OFF, ...(opts.focus ?? shot.focus) }, opts.focusMs ?? ms, opts.focusEase ?? opts.ease ?? 'move');
  if (shot.tilt || opts.tilt) M.tween('tilt', opts.tilt ?? shot.tilt, ms, opts.ease ?? 'move');
  if (shot.grade || opts.grade) M.tween('grade', opts.grade ?? shot.grade, ms, opts.ease ?? 'move');
  M.follow = true;
  return shot;
};
/** A hard cut. */
M.cut = (s, opts = {}) => {
  const shot = shotOf(s);
  if (shot.enter) { try { shot.enter(); } catch (e) { M.errs.push('enter ' + e); } }
  M.set('cam', shot);
  if (shot.focus || opts.focus) M.set('focus', { ...M.FOCUS_OFF, ...(opts.focus ?? shot.focus) });
  if (shot.tilt || opts.tilt) M.set('tilt', opts.tilt ?? shot.tilt);
  if (shot.grade || opts.grade) M.set('grade', opts.grade ?? shot.grade);
  M.follow = true;
  if (opts.onCut) opts.onCut();
  return shot;
};
/**
 * A whip pan: the camera swings to the next shot in `ms` with a horizontal motion blur that peaks at
 * mid-swing; `swap` runs on the frame the blur is at its peak (the hidden cut for a painting swap).
 */
M.whip = (s, opts = {}) => {
  const shot = shotOf(s);
  const ms = opts.ms ?? 300;
  M.tween('cam', shot, ms, opts.ease ?? 'whip');
  if (shot.focus || opts.focus) M.tween('focus', { ...M.FOCUS_OFF, ...(opts.focus ?? shot.focus) }, ms, 'inout');
  if (shot.tilt || opts.tilt) M.tween('tilt', opts.tilt ?? shot.tilt, ms, 'inout');
  M.whipFx = { t0: M.t, dur: ms / 1000, peak: opts.peak ?? 48, dir: opts.dir ?? 1, swap: opts.swap, swapped: false, vertical: opts.vertical ?? 0 };
  M.follow = true;
  return shot;
};
M.apply = (dt) => {
  const c = M.cam();
  const v = M.val('cam');
  if (!v) return;
  M.sway.clock += dt;
  const a = M.sway.amp * (v.sway ?? 1), s = M.sway.speed, k = M.sway.clock;
  const ox = Math.sin(k * s) * a + Math.sin(k * s * 0.37 + 1.1) * a * 0.4;
  const oy = Math.sin(k * s * 0.73 + 2.2) * a * 0.55;
  const oz = Math.sin(k * s * 0.51 + 0.4) * a * 0.3;
  const sh = M.shakeNow ? M.shakeNow(dt) : [0, 0];
  c.position.set(v.pos[0] + ox + sh[0], v.pos[1] + oy + sh[1], v.pos[2] + oz);
  c.lookAt(v.look[0] + ox * 0.18 + sh[0] * 0.5, v.look[1] + oy * 0.18 + sh[1] * 0.5, v.look[2]);
  if (v.roll) c.rotateZ((v.roll * Math.PI) / 180);
  if (v.fov && Math.abs(c.fov - v.fov) > 1e-4) { c.fov = v.fov; c.updateProjectionMatrix(); }
  c.updateMatrixWorld(true);
};
// A small screen shake of our own (the game's shake lives on the rig we have taken over).
M.shake = (amp = 0.08, ms = 260) => { M._shake = { amp, left: ms / 1000, total: ms / 1000, ph: 0 }; };
M.shakeNow = (dt) => {
  const s = M._shake; if (!s || s.left <= 0) return [0, 0];
  s.left -= dt; s.ph += 0.9; const k = Math.max(0, s.left / s.total) ** 2;
  return [Math.sin(s.ph * 2.7) * s.amp * k, Math.sin(s.ph * 3.9 + 1.7) * s.amp * k * 0.8];
};

// ------------------------------------------------------------------ post passes (focus + whip)
M.FOCUS_OFF = { id: null, pad: 1.25, cx: 0.5, cy: 0.5, rx: 0.5, ry: 0.5, blur: 0, feather: 0.6 };
const VERT = 'varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0); }';
const FOCUS_FRAG = `
uniform sampler2D tDiffuse; uniform vec2 resolution; uniform vec2 direction; uniform vec2 center; uniform vec2 radii;
uniform float feather; uniform float maxBlur; varying vec2 vUv;
void main(){
  vec4 base = texture2D(tDiffuse, vUv);
  float d = length((vUv - center) / max(radii, vec2(0.0001)));
  float r = maxBlur * smoothstep(1.0, 1.0 + feather, d);
  if (r < 0.05) { gl_FragColor = base; return; }
  vec2 st = direction * resolution * (r / 5.0);
  vec4 s = base * 0.1592;
  s += (texture2D(tDiffuse, vUv + st * 1.0) + texture2D(tDiffuse, vUv - st * 1.0)) * 0.1476;
  s += (texture2D(tDiffuse, vUv + st * 2.0) + texture2D(tDiffuse, vUv - st * 2.0)) * 0.1177;
  s += (texture2D(tDiffuse, vUv + st * 3.0) + texture2D(tDiffuse, vUv - st * 3.0)) * 0.0806;
  s += (texture2D(tDiffuse, vUv + st * 4.0) + texture2D(tDiffuse, vUv - st * 4.0)) * 0.0474;
  s += (texture2D(tDiffuse, vUv + st * 5.0) + texture2D(tDiffuse, vUv - st * 5.0)) * 0.0271;
  gl_FragColor = s / (0.1592 + 2.0 * (0.1476 + 0.1177 + 0.0806 + 0.0474 + 0.0271));
}`;
const WHIP_FRAG = `
uniform sampler2D tDiffuse; uniform vec2 resolution; uniform vec2 direction; uniform float amount; varying vec2 vUv;
void main(){
  if (amount < 0.1) { gl_FragColor = texture2D(tDiffuse, vUv); return; }
  vec2 st = direction * resolution * (amount / 8.0);
  vec4 s = vec4(0.0);
  for (int i = -8; i <= 8; i++) { s += texture2D(tDiffuse, vUv + st * float(i)); }
  gl_FragColor = s / 17.0;
}`;
M.passes = () => {
  const R = M.R();
  if (M._fp && R.composer.passes.includes(M._fp.h)) return M._fp;
  const SP = R.tiltH.constructor;
  const V2 = R.tiltH.uniforms.resolution.value.constructor;
  const mk = (frag, extra) => ({ name: 'Motion', uniforms: { tDiffuse: { value: null }, resolution: { value: new V2(1 / 1600, 1 / 900) }, direction: { value: new V2(1, 0) }, ...extra }, vertexShader: VERT, fragmentShader: frag });
  const focusU = () => ({ center: { value: new V2(0.5, 0.5) }, radii: { value: new V2(0.5, 0.5) }, feather: { value: 0.6 }, maxBlur: { value: 0 } });
  const h = new SP(mk(FOCUS_FRAG, focusU()));
  const v = new SP(mk(FOCUS_FRAG, focusU()));
  v.uniforms.direction.value.set(0, 1);
  const w = new SP(mk(WHIP_FRAG, { amount: { value: 0 } }));
  const i = R.composer.passes.indexOf(R.gradePass);
  R.composer.insertPass(h, i);
  R.composer.insertPass(v, i + 1);
  R.composer.insertPass(w, i + 2);
  M._fp = { h, v, w };
  return M._fp;
};
M.unpasses = () => { const R = M.R(); if (!M._fp) return; for (const p of Object.values(M._fp)) R.composer.removePass(p); M._fp = null; };
/** Resolve a focus value (possibly following an actor) to an ellipse in UV (0..1, y up). */
M.focusEllipse = (f) => {
  if (!f) return null;
  let { cx, cy, rx, ry } = f;
  if (f.id) {
    const r = M.st().projectRect(f.id);
    // Measured against the canvas, not the window: on a phone the canvas is wider than the page and offset.
    const cv = M.R().domElement.getBoundingClientRect();
    const W = cv.width, H = cv.height;
    if (r) {
      const pad = f.pad ?? 1.25;
      cx = (r.x - cv.x + r.w / 2) / W; cy = 1 - (r.y - cv.y + r.h * (f.yBias ?? 0.5)) / H;
      rx = (r.w / 2 / W) * pad * (f.xPad ?? 1); ry = (r.h / 2 / H) * pad;
    }
  }
  return { cx, cy, rx, ry, blur: f.blur ?? 0, feather: f.feather ?? 0.6 };
};
/** The focus ellipse right now, both ends resolved so a follow tracks its actor through the move. */
M.focusNow = () => {
  const c = M.ch.focus;
  if (!c) return null;
  if (!c.dur) return M.focusEllipse(c.to);
  const k = Math.min(1, Math.max(0, (M.t - c.t0) / c.dur));
  return lerpAny(M.focusEllipse(c.from), M.focusEllipse(c.to), ease(c.ease)(k));
};
M.applyFx = (dt) => {
  const R = M.R();
  const c = M.ch.focus;
  if (c) {
    const fp = M.passes();
    const e = M.focusNow();
    const res = R.tiltH.uniforms.resolution.value;
    for (const pass of [fp.h, fp.v]) {
      pass.uniforms.resolution.value.copy(res);
      pass.uniforms.center.value.set(e.cx, e.cy);
      pass.uniforms.radii.value.set(Math.max(0.001, e.rx), Math.max(0.001, e.ry));
      pass.uniforms.feather.value = e.feather;
      pass.uniforms.maxBlur.value = e.blur;
    }
  }
  const tl = M.val('tilt');
  if (tl) R.applyPost({ tiltFocus: tl.focus, tiltBandWidth: tl.band, tiltMaxBlur: tl.blur });
  const g = M.val('grade');
  if (g) { if (g.vignette != null) R.gradePass.uniforms.vignette.value = g.vignette; if (g.vignetteRadius != null) R.gradePass.uniforms.vignetteRadius.value = g.vignetteRadius; if (g.exposure != null) R.renderer.toneMappingExposure = g.exposure; if (g.saturation != null) R.gradePass.uniforms.saturation.value = g.saturation; }
  const wf = M.whipFx;
  if (wf) {
    const fp = M.passes();
    const k = (M.t - wf.t0) / wf.dur;
    const amt = k <= 0 || k >= 1 ? 0 : wf.peak * Math.pow(Math.sin(Math.PI * k), 1.6);
    fp.w.uniforms.resolution.value.copy(R.tiltH.uniforms.resolution.value);
    fp.w.uniforms.direction.value.set(wf.vertical ? 0 : 1, wf.vertical ? 1 : 0);
    fp.w.uniforms.amount.value = amt;
    if (!wf.swapped && k >= 0.5) { wf.swapped = true; try { wf.swap?.(); } catch (e) { M.errs.push('swap ' + e); } }
    if (k >= 1) M.whipFx = null;
  }
};

// ------------------------------------------------------------------ actors
M.faces = M.faces || {};
M.rot0 = M.rot0 || {};
/** Turn an actor's plane toward the lens (k = 1 square on, 0.5 halfway, null to hand it back). */
M.faceLens = (id, k = 1) => {
  const a = M.act(id); if (!a) return 'no ' + id;
  if (!(id in M.rot0)) M.rot0[id] = a.rotation.y;
  if (k == null) { delete M.faces[id]; a.rotation.y = M.rot0[id]; return 'off'; }
  M.faces[id] = k; return k;
};
M.faceTick = () => {
  const c = M.cam();
  for (const [id, k] of Object.entries(M.faces)) {
    const a = M.act(id); if (!a) continue;
    const az = Math.atan2(c.position.x - a.position.x, c.position.z - a.position.z);
    a.rotation.y = (M.rot0[id] ?? 0) + az * k;
  }
};
/** Planes' angle off the lens in degrees, per actor (the cardboard meter). */
M.offLens = () => {
  const c = M.cam(); const o = {};
  for (const id of M.st().staged()) {
    const a = M.act(id); if (!a) continue;
    const az = (Math.atan2(c.position.x - a.position.x, c.position.z - a.position.z) * 180) / Math.PI;
    const yaw = ((a.rotation.y + (a.inner?.rotation.y ?? 0)) * 180) / Math.PI;
    let d = az - yaw; while (d > 180) d -= 360; while (d < -180) d += 360;
    o[id] = +d.toFixed(1);
  }
  return o;
};

/** A stand-in plate behind the field for reverse angles: today's backdrop, mirrored and softened. */
M.plate = (on = true, opts = {}) => {
  const sc = M.st().opts.scene;
  let m = sc.getObjectByName('motion-reverse-plate');
  if (!on) { if (m) m.visible = false; return false; }
  if (!m) {
    const bp = sc.getObjectByName('backdrop-painting');
    const tex = bp.material.map;
    const img = tex.image;
    const cv = document.createElement('canvas');
    cv.width = Math.round((img.width || img.naturalWidth) / 2); cv.height = Math.round((img.height || img.naturalHeight) / 2);
    const g = cv.getContext('2d'); g.filter = `blur(${opts.blur ?? 5}px)`; g.drawImage(img, 0, 0, cv.width, cv.height);
    const t2 = new tex.constructor(cv); t2.colorSpace = tex.colorSpace; t2.flipY = tex.flipY; t2.needsUpdate = true;
    const mat = bp.material.clone(); mat.map = t2;
    m = new bp.constructor(bp.geometry, mat);
    m.name = 'motion-reverse-plate';
    m.renderOrder = bp.renderOrder;
    sc.add(m);
  }
  const bp = sc.getObjectByName('backdrop-painting');
  const bw = new bp.position.constructor(); bp.getWorldPosition(bw);
  m.position.set(bw.x + (opts.dx ?? 0), bw.y + (opts.dy ?? 0), opts.z ?? -bw.z);
  m.rotation.set(0, Math.PI, 0);
  const s = opts.scale ?? 1;
  m.scale.set(-s, s, 1);
  m.visible = true;
  return m.position.toArray();
};

/** A named stand-in plate: a copy of today's backdrop painting aimed at a camera (softened, optionally mirrored). */
M.plateAt = (name, opts = {}) => {
  const sc = M.st().opts.scene;
  let m = sc.getObjectByName('motion-plate-' + name);
  if (opts.off) { if (m) m.visible = false; return false; }
  if (!m) {
    const bp = sc.getObjectByName('backdrop-painting');
    const tex = bp.material.map; const img = tex.image;
    const cv = document.createElement('canvas');
    cv.width = Math.round((img.width || img.naturalWidth) / 2); cv.height = Math.round((img.height || img.naturalHeight) / 2);
    const g = cv.getContext('2d'); g.filter = `blur(${opts.blur ?? 4}px)`; g.drawImage(img, 0, 0, cv.width, cv.height);
    const t2 = new tex.constructor(cv); t2.colorSpace = tex.colorSpace; t2.flipY = tex.flipY; t2.needsUpdate = true;
    const mat = bp.material.clone(); mat.map = t2;
    m = new bp.constructor(bp.geometry, mat); m.name = 'motion-plate-' + name; m.renderOrder = bp.renderOrder; sc.add(m);
  }
  m.position.set(...opts.pos); m.rotation.set(0, opts.yaw ?? 0, 0);
  const k = opts.scale ?? 1; m.scale.set(opts.mirror ? -k : k, k, 1);
  m.visible = true; return m.position.toArray();
};
/** Aim a plate at the far end of a shot's view axis, `dist` units out. */
M.plateFor = (name, shot, opts = {}) => {
  const s = shotOf(shot); const d = [s.look[0] - s.pos[0], 0, s.look[2] - s.pos[2]]; const n = Math.hypot(d[0], d[2]); d[0] /= n; d[2] /= n;
  const dist = opts.dist ?? 64; const y = opts.y ?? -8;
  return M.plateAt(name, { pos: [s.pos[0] + d[0] * dist, y, s.pos[2] + d[2] * dist], yaw: Math.atan2(-d[0], -d[2]), mirror: opts.mirror, blur: opts.blur, scale: opts.scale });
};
M.platesOff = () => { const sc = M.st().opts.scene; sc.traverse((n) => { if (n.name && n.name.startsWith('motion-plate-')) n.visible = false; }); };

M.hookEvents = () => {
  const hud = M.hud(); if (!hud) return false;
  if (!hud.__origOnEvent) hud.__origOnEvent = hud.onEvent.bind(hud);
  hud.onEvent = (e) => { try { (window.__evlog || (window.__evlog = [])).push({ t: +performance.now().toFixed(0), type: e.type, actor: e.actorId || e.sourceId || '', name: e.abilityName || e.name || e.status || '' }); M.onEv(e); } catch (err) {} return hud.__origOnEvent(e); };
  return true;
};

/** Re-project the target reticle (FFX HUD) — the live HUD only does this on a selection change. */
/** Record every TargetCursor the HUDs create (FFX's is a field, FFX-2's a closure), so the mock can re-project it. */
M.patchCursor = async () => {
  const mod = await import('/src/ui/ffx/TargetCursor.ts');
  const proto = mod.TargetCursor.prototype;
  if (!proto.__motionPatched) {
    for (const k of ['showSingle', 'showGroup', 'step', 'reposition']) {
      const orig = proto[k];
      if (typeof orig !== 'function') continue;
      proto[k] = function (...a) { (window.__tcs || (window.__tcs = new Set())).add(this); return orig.apply(this, a); };
    }
    proto.__motionPatched = true;
  }
  const h = M.hud(); const tc = h?.commandMenu?.targetCursor; if (tc) (window.__tcs || (window.__tcs = new Set())).add(tc);
  return true;
};
/** Re-project the live target reticle: the HUD itself only does this on a selection change. */
M.reticle = () => {
  let n = 0;
  try {
    for (const tc of window.__tcs || []) {
      if (!tc.el?.isConnected) { window.__tcs.delete(tc); continue; }
      if (tc.entries?.length) { tc.reposition(); n++; }
    }
  } catch (e) { M.errs.push('reticle ' + e); }
  return n;
};
M.hudRoot = () => document.querySelector('[data-screen="battle"]');
M.hudOpacity = (k) => { const r = M.hudRoot(); if (!r) return false; for (const el of r.children) if (el.tagName !== 'CANVAS') el.style.opacity = String(k); return true; };

// ------------------------------------------------------------------ the per-frame hook
M.tick = (dt) => {
  M.t += dt;
  if (M.on) { M.shakeFollow(); M.apply(dt); }
  M.faceTick();
  M.applyFx(dt);
  if (M.follow) {
    const moving = M.moving('cam') || !!M.whipFx;
    if (moving || M._wasMoving) M.reticle();
    M._wasMoving = moving;
  }
  if (M.onTick) { try { M.onTick(dt); } catch (e) { M.errs.push('onTick ' + e); } }
};
M.hook = () => {
  const bc = M.bc();
  if (bc.__motionHooked) return false;
  const proto = Object.getPrototypeOf(bc);
  bc.update = function (dt) { proto.update.call(this, dt); try { M.tick(dt); } catch (e) { if (M.errs.length < 30) M.errs.push(String(e && e.stack || e).slice(0, 300)); } };
  bc.__motionHooked = true;
  return true;
};
