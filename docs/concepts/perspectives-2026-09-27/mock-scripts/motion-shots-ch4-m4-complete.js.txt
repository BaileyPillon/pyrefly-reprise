// Chapter IV M4 complete: Paine's command over her shoulder, then one camera per beat with the PAINTED reverse plate
// and Bahamut's rear painting on his turns.
const M = __motion;
const PARTY = ['yuna', 'rikku', 'paine'];
const MIST = ['mist-0', 'mist-1', 'mist-2', 'mist-3'];
const clean = () => { M.frontOn(PARTY); M.showFront('bahamut'); for (const id of Object.keys(M.faces)) M.faceLens(id, null); for (const id of Object.keys(M.sil)) M.unsilhouette(id); M.sil = {}; for (const id of [...PARTY, 'bahamut']) M.rim(id, null); M.act('bahamut').setBrightness(1); M.platesAllOff(); M.setVis(MIST, true); };
M.shots.m4_cmd = { ...M.shots.ots_atk, enter() { clean(); M.shots.ots_atk.enter(); } };
M.shots.side4 = { pos: [4, 0.95, 14], look: [0.4, 1.6, -2.5], fov: 24, roll: 0, sway: 0.5, focus: { ...M.FOCUS_OFF }, tilt: { focus: 0.5, band: 0.22, blur: 3.0 }, enter() { clean(); } };
M.shots.rev4 = { pos: [8.5, 1.8, -9.5], look: [-1.6, 1.2, 0.2], fov: 24, roll: 0, sway: 0.5, focus: { ...M.FOCUS_OFF }, tilt: { focus: 0.5, band: 0.18, blur: 3.0 },
  enter() { clean(); M.setVis(MIST, false); for (const id of PARTY) M.faceLens(id, 1); M.showRear('bahamut', 'rear34'); M.faceLens('bahamut', 1);
    M.rim('bahamut', { rim: 0xffb0d0, rimStrength: 1.3, rimDir: [-0.6, 1], rimWidth: 3.4 }); M.act('bahamut').setBrightness(0.7);
    M.placePlate('rev', M.shots.rev4, { dist: 60, horizon: 0.45 }); } };
M.shots.colo4 = { pos: [0.2, 0.3, -1.6], look: [1.05, 2.8, -5.8], fov: 40, roll: 0, sway: 0.4, focus: { ...M.FOCUS_OFF }, tilt: { focus: 0.55, band: 0.3, blur: 2.0 }, enter() { clean(); M.setVis(MIST, false); } };
M.shots.hero4 = { pos: [-1.2, 0.5, 5.2], look: [-1.3, 1.45, -0.2], fov: 34, roll: 0, sway: 0.6, focus: { ...M.FOCUS_OFF }, tilt: { focus: 0.5, band: 0.3, blur: 2.0 }, enter() { clean(); } };
M.m4clean = clean;
return ['m4_cmd', 'side4', 'rev4', 'colo4', 'hero4'];
