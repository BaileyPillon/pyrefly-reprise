// Chapter IV Director's Cut shots (M4), Bevelle Underground. World units. Stand-in plates are today's backdrop re-aimed (tagged).
const M = __motion;
await M.patchCursor();
const ALL = () => M.st().staged();
const PARTY = ['yuna', 'rikku', 'paine'];
const BOSS = ['bahamut'];
const MIST = ['mist-0', 'mist-1', 'mist-2', 'mist-3'];
const clean = () => { for (const id of Object.keys(M.faces)) M.faceLens(id, null); for (const id of Object.keys(M.sil)) M.unsilhouette(id); M.sil = {}; M.platesOff(); M.setVis(MIST, true); };
const noMist = () => M.setVis(MIST, false);
M.shots.side = { pos: [9.0, 0.9, 7.0], look: [-0.3, 1.5, -2.5], fov: 26, roll: 0, sway: 0.5, focus: { ...M.FOCUS_OFF }, tilt: { focus: 0.5, band: 0.2, blur: 3.0 },
  enter() { clean(); noMist(); for (const id of ALL()) M.faceLens(id, 1); M.plateFor('side', M.shots.side, { dist: 70, y: -3, blur: 3 }); } };
M.shots.rev = { pos: [8.5, 1.8, -9.5], look: [-1.6, 1.2, 0.2], fov: 24, roll: 0, sway: 0.5, focus: { ...M.FOCUS_OFF }, tilt: { focus: 0.5, band: 0.18, blur: 3.5 },
  enter() { clean(); noMist(); for (const id of PARTY) M.faceLens(id, 1); for (const id of BOSS) M.silhouette(id, 0.05, '#ffb0d0', 1.8); M.plateFor('rev', M.shots.rev, { dist: 62, y: -3, mirror: true, blur: 7 }); } };
M.shots.colo = { pos: [0.2, 0.3, -1.6], look: [1.05, 2.8, -5.8], fov: 40, roll: 0, sway: 0.4, focus: { ...M.FOCUS_OFF }, tilt: { focus: 0.55, band: 0.3, blur: 2.0 },
  enter() { clean(); noMist(); } };
M.shots.hero = { pos: [-1.2, 0.5, 5.2], look: [-1.3, 1.45, -0.2], fov: 34, roll: 0, sway: 0.6, focus: { ...M.FOCUS_OFF }, tilt: { focus: 0.5, band: 0.3, blur: 2.0 },
  enter() { clean(); } };
return Object.keys(M.shots).length;
