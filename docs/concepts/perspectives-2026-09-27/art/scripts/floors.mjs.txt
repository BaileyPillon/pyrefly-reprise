#!/usr/bin/env node
/**
 * Perspectives round 2026-09-27 (art agent, stage 2 item 7): paint a diorama floor plate over its layout
 * sketch (floors.py sketch), 1024x1024 img2img -> RealESRGAN x4 = 4096x4096, then floors.py finish (the
 * falloff to the fog colour over the outer 10 %). Candidates only.
 *
 *   node tools/zz-persp-art/floors.mjs <gagazet|bevelle> <n> <seed> <denoise>
 */
import { mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { join, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
import { spawnSync } from 'node:child_process';
import { stageImage, BASE_NEGATIVE, QUALITY_TAGS, joinTags } from '../gen/comfy.mjs';
import { maxRgbOfPng, isBlackFrame } from '../gen/black-frame.mjs';

const CAND = 'D:/Tools/pyrefly-art-backup/candidates/2026-09-27-perspectives/floors';
const SCR = 'D:/Tools/pyrefly-scratch/persp-art/floors';
const PY = 'D:/Tools/ComfyUI/python_embeded/python.exe';
const HERE = fileURLToPath(new URL('.', import.meta.url));
const COMFY = process.env.COMFY_URL || 'http://127.0.0.1:8188';
const CKPT = 'animagine-xl-4.0-opt.safetensors';
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

const TAGS = {
  gagazet: 'no humans, from above, top-down view, overhead view, ground texture, snowfield at night, wind-carved snow drifts, cracked '
    + 'blue ice sheets, dark rocks poking through the snow, footprints in the snow, tiny glowing specks, cold blue moonlight, painterly, detailed',
  bevelle2: 'no humans, from above, top-down view, overhead view, metal floor texture, riveted steel deck plates, grating, (torn hole in the '
    + 'metal floor:1.3), (broken jagged steel plates around the hole:1.2), bright cyan light shining up out of the hole, cyan light lines in the '
    + 'floor seams, small brass inlay, oil stains, warm orange lamp light at the edges, dark industrial machina floor, painterly, detailed',
  bevelle: 'no humans, from above, top-down view, overhead view, metal floor texture, riveted steel deck plates, grating, cyan light '
    + 'lines in the floor seams, a torn hole in the floor with cyan light shining up from below, brass ring inlay in the floor, oil stains, '
    + 'warm orange lamp light at the edges, dark industrial machina floor, painterly, detailed',
};
const NEG = 'sky, horizon, mountains, perspective, vanishing point, walls, pillars, people, character, 1girl, 1boy, text, letters, '
  + 'border, frame, trees, buildings, grass, sun';

function graph({ init, positive, negative, seed, denoise, prefix }) {
  return {
    4: { class_type: 'CheckpointLoaderSimple', inputs: { ckpt_name: CKPT } },
    6: { class_type: 'CLIPTextEncode', inputs: { text: positive, clip: ['4', 1] } },
    7: { class_type: 'CLIPTextEncode', inputs: { text: negative, clip: ['4', 1] } },
    30: { class_type: 'LoadImage', inputs: { image: init, upload: 'image' } },
    32: { class_type: 'VAEEncode', inputs: { pixels: ['30', 0], vae: ['4', 2] } },
    3: { class_type: 'KSampler', inputs: { seed, steps: 30, cfg: 6, sampler_name: 'euler_ancestral', scheduler: 'normal', denoise, model: ['4', 0], positive: ['6', 0], negative: ['7', 0], latent_image: ['32', 0] } },
    8: { class_type: 'VAEDecode', inputs: { samples: ['3', 0], vae: ['4', 2] } },
    10: { class_type: 'UpscaleModelLoader', inputs: { model_name: 'RealESRGAN_x4plus.pth' } },
    11: { class_type: 'ImageUpscaleWithModel', inputs: { upscale_model: ['10', 0], image: ['8', 0] } },
    9: { class_type: 'SaveImage', inputs: { filename_prefix: prefix, images: ['11', 0] } },
  };
}

/** The hole pass: a masked img2img of the hole region only; outside the mask the deck stays pixel-identical. */
function holeGraph({ init, mask, positive, negative, seed, denoise, prefix }) {
  return {
    4: { class_type: 'CheckpointLoaderSimple', inputs: { ckpt_name: CKPT } },
    6: { class_type: 'CLIPTextEncode', inputs: { text: positive, clip: ['4', 1] } },
    7: { class_type: 'CLIPTextEncode', inputs: { text: negative, clip: ['4', 1] } },
    30: { class_type: 'LoadImage', inputs: { image: init, upload: 'image' } },
    31: { class_type: 'LoadImageMask', inputs: { image: mask, channel: 'red', upload: 'image' } },
    32: { class_type: 'VAEEncode', inputs: { pixels: ['30', 0], vae: ['4', 2] } },
    33: { class_type: 'SetLatentNoiseMask', inputs: { samples: ['32', 0], mask: ['31', 0] } },
    3: { class_type: 'KSampler', inputs: { seed, steps: 30, cfg: 6, sampler_name: 'euler_ancestral', scheduler: 'normal', denoise, model: ['4', 0], positive: ['6', 0], negative: ['7', 0], latent_image: ['33', 0] } },
    8: { class_type: 'VAEDecode', inputs: { samples: ['3', 0], vae: ['4', 2] } },
    34: { class_type: 'ImageCompositeMasked', inputs: { destination: ['30', 0], source: ['8', 0], x: 0, y: 0, resize_source: false, mask: ['31', 0] } },
    10: { class_type: 'UpscaleModelLoader', inputs: { model_name: 'RealESRGAN_x4plus.pth' } },
    11: { class_type: 'ImageUpscaleWithModel', inputs: { upscale_model: ['10', 0], image: ['34', 0] } },
    9: { class_type: 'SaveImage', inputs: { filename_prefix: prefix, images: ['11', 0] } },
  };
}
const HOLE_TAGS = 'no humans, from above, top-down view, overhead view, metal floor texture, (torn hole in the metal floor:1.3), jagged '
  + 'broken steel plates bent down into the hole, bright cyan light shining up from below, riveted steel deck plates, dark industrial, painterly, detailed';

async function main() {
  const [name, nArg, seedArg, denArg] = process.argv.slice(2);
  const holeFrom = process.argv.includes('--hole') ? process.argv[process.argv.indexOf('--hole') + 1] : null;
  const tagKey = process.argv.includes('--tags') ? process.argv[process.argv.indexOf('--tags') + 1] : name;
  if (!TAGS[tagKey]) throw new Error('floor must be gagazet or bevelle');
  const n = Number(nArg); const seed = Number(seedArg); const denoise = Number(denArg);
  mkdirSync(join(CAND, name), { recursive: true });
  const sketchName = process.argv.includes('--sketch') ? process.argv[process.argv.indexOf('--sketch') + 1] : `sketch-${name}.png`;
  let init = stageImage(join(SCR, sketchName));
  let maskName = null;
  if (holeFrom) {
    const hi = join(SCR, `hole-init-${nArg}.png`); const hm = join(SCR, 'hole-mask.png');
    const rr = spawnSync(PY, ['-s', join(HERE, 'floors.py'), 'holeinit', join(CAND, name, `${holeFrom}.painted.png`), hi, hm], { encoding: 'utf8' });
    if (rr.status !== 0) throw new Error(rr.stderr);
    init = stageImage(hi); maskName = stageImage(hm);
  }
  const positive = joinTags(holeFrom ? HOLE_TAGS : TAGS[tagKey], QUALITY_TAGS);
  const negative = joinTags(BASE_NEGATIVE, NEG);
  for (;;) { const q = await (await fetch(`${COMFY}/queue`)).json(); if ((q.queue_pending?.length || 0) < 3) break; await sleep(4000); }
  const t0 = Date.now();
  const res = await fetch(`${COMFY}/prompt`, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ prompt: holeFrom ? holeGraph({ init, mask: maskName, positive, negative, seed, denoise, prefix: `pyrefly/persp-2026-09-27/floor-${name}-hole` }) : graph({ init, positive, negative, seed, denoise, prefix: `pyrefly/persp-2026-09-27/floor-${name}` }), client_id: 'pyrefly-persp-art' }) });
  const body = await res.json();
  if (!res.ok || body.error) throw new Error(`ComfyUI rejected: ${JSON.stringify(body).slice(0, 1500)}`);
  let e;
  for (;;) {
    await sleep(2500);
    e = (await (await fetch(`${COMFY}/history/${body.prompt_id}`)).json())[body.prompt_id];
    if (e?.status?.status_str === 'error') throw new Error(JSON.stringify(e.status.messages).slice(0, 800));
    if (e?.outputs?.['9']?.images?.length) break;
  }
  const img = e.outputs['9'].images[0];
  const qs = new URLSearchParams({ filename: img.filename, subfolder: img.subfolder || '', type: img.type || 'output' });
  const painted = join(CAND, name, `cand-${n}.painted.png`);
  writeFileSync(painted, Buffer.from(await (await fetch(`${COMFY}/view?${qs}`)).arrayBuffer()));
  const maxRgb = maxRgbOfPng(readFileSync(painted));
  if (maxRgb != null && isBlackFrame(maxRgb)) throw new Error(`${name} #${n}: BLACK FRAME, stopping (no re-roll)`);
  const r = spawnSync(PY, ['-s', join(HERE, 'floors.py'), 'finish', name, painted, join(CAND, name, `cand-${n}.png`)], { encoding: 'utf8' });
  if (r.status !== 0) throw new Error(r.stderr);
  writeFileSync(join(CAND, name, `cand-${n}.json`), JSON.stringify({
    round: 'perspectives-2026-09-27', subject: `floor-${name}`, view: 'top', candidate: n, seed, denoise, positive, negative,
    sketch: sketchName, tagSet: tagKey,
    init: holeFrom ? `${holeFrom} downscaled to 1024 with a torn-hole sketch pasted over the hole (floors.py holeinit); masked img2img of the hole region only (SetLatentNoiseMask + ImageCompositeMasked)` : `layout sketch (tools/zz-persp-art/floors.py sketch ${name}), 1024x1024`, model: CKPT, steps: 30, cfg: 6,
    sampler: 'euler_ancestral', scheduler: 'normal', upscaler: 'RealESRGAN_x4plus.pth (x4, 1024 -> 4096)', finish: r.stdout.trim(),
    seconds: (Date.now() - t0) / 1000, generatedAt: new Date().toISOString(),
  }, null, 1));
  console.log(`[persp-floor] ${name} #${n} seed ${seed} d${denoise} ${((Date.now() - t0) / 1000).toFixed(0)} s`);
}

const isMain = process.argv[1] && resolve(process.argv[1]) === fileURLToPath(import.meta.url);
if (isMain) await main();
