def font(size):
    for f in ("arialbd.ttf", "arial.ttf"):
        try:
            return ImageFont.truetype(f, size)
        except Exception:
            pass
    return ImageFont.load_default()


def tile(path, label, sub, pick=False):
    im = Image.open(path).convert("RGBA")
    s = TILE_H / im.height
    im = im.resize((max(1, round(im.width * s)), TILE_H), Image.LANCZOS)
    pad = 6
    t = Image.new("RGB", (im.width + 2 * pad, TILE_H + 2 * pad + 44), (34, 34, 38))
    ground = Image.new("RGBA", im.size, (120, 120, 124, 255))
    ground.alpha_composite(im)
    t.paste(ground.convert("RGB"), (pad, pad + 44))
    d = ImageDraw.Draw(t)
    if pick:
        d.rectangle([0, 0, t.width - 1, t.height - 1], outline=GOLD, width=5)
    d.text((pad + 2, 4), label, fill=GOLD if pick else (235, 235, 235), font=font(17))
    d.text((pad + 2, 24), sub, fill=GOLD if pick else (175, 175, 175), font=font(14))
    return t


def main():
    OUT.mkdir(parents=True, exist_ok=True)
    for nn, subj, pick, verdicts, why in SUBJECTS:
        tiles = [tile(IDLES / subj / "idle.png", "IDLE (front)", "shipped, public/art")]
        for arm_dir in sorted(p for p in (CAND / subj).iterdir() if p.is_dir()):
            for cj in sorted(arm_dir.glob("cand-*.json"), key=lambda p: int(p.stem.split("-")[1])):
                meta = json.loads(cj.read_text(encoding="utf-8"))
                key = f"{arm_dir.name}/{meta['candidate']}"
                is_pick = f"{arm_dir.name}/{cj.stem}" == pick
                path = (CAND / subj / "pick-rear34.png") if is_pick else cj.with_suffix(".png")
                label = f"{'PICK  ' if is_pick else ''}{arm_dir.name} #{meta['candidate']}  s{meta['seed']}"
                tiles.append(tile(path, label, "installed rear34.png" if is_pick else verdicts.get(key, ""), is_pick))
        rows = [tiles[i:i + PER_ROW] for i in range(0, len(tiles), PER_ROW)]
        W = max(sum(t.width for t in r) + 10 * (len(r) + 1) for r in rows)
        rh = tiles[0].height
        head = 52
        foot = 70
        H = head + len(rows) * (rh + 10) + foot
        sheet = Image.new("RGB", (W, H), (22, 22, 26))
        d = ImageDraw.Draw(sheet)
        d.text((12, 12), f"{subj}: rear three-quarter candidates, every render (perspectives round 2026-09-27, art agent)",
               fill=(240, 240, 240), font=font(24))
        y = head
        for r in rows:
            x = 10
            for t in r:
                sheet.paste(t, (x, y))
                x += t.width + 10
            y += rh + 10
        import textwrap
        chars = max(60, int(W / 10.5))
        for i, line in enumerate(textwrap.wrap(why, chars)[:2]):
            d.text((12, y + 8 + 28 * i), line, fill=GOLD, font=font(18))
        out = OUT / f"sheet-{nn}-{subj}.jpg"
        sheet.save(out, quality=88)
        print(out.name, sheet.size, len(tiles) - 1, "candidates")


if __name__ == "__main__":
    main()
