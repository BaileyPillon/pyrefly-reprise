export const ARMS = {
  A: { cn: null },
  B: { cn: [0.75, 0.85, 0.8, 0.8, 0.75, 0.85] },
};

/** Variant knobs (CLI): pose override, identity words dropped/added, raw emphasis, extra negatives. */
export function promptFor(sub, v = {}) {
  const s = SUBJECTS[sub];
  const drop = new Set((v.idDrop || '').split('|').map((x) => x.trim()).filter(Boolean));
  let identity = s.identity.split(', ').filter((x) => !drop.has(x)).join(', ');
  if (v.idAdd) identity = `${identity}, ${v.idAdd}`;
  const lint = lintSpritePrompt({ tags: identity, poseTags: v.pose || s.pose, composition: s.framing ? 'boss' : 'full' });
  if (lint.stripped.length) throw new Error(`lint stripped ${JSON.stringify(lint.stripped)}`);
  return [escapeTags(lint.tags), escapeTags(lint.poseTags), v.emph || s.emph || '', REAR_VIEW, s.framing || FRAMING, STYLE_TAGS, QUALITY_TAGS]
    .filter(Boolean).join(', ');
}
export function negativeFor(sub, v = {}) {
  return [SPRITE_NEGATIVE, REAR_NEGATIVE, SUBJECTS[sub].neg, v.negAdd || ''].filter(Boolean).join(', ');
}

function graph({ positive, negative, seed, cn, skel, refA, refB, ipa, prefix, width = 832, height = 1216 }) {
  const g = {
    4: { class_type: 'CheckpointLoaderSimple', inputs: { ckpt_name: CKPT } },
    5: { class_type: 'EmptyLatentImage', inputs: { width, height, batch_size: 1 } },
    6: { class_type: 'CLIPTextEncode', inputs: { text: positive, clip: ['4', 1] } },
    7: { class_type: 'CLIPTextEncode', inputs: { text: negative, clip: ['4', 1] } },
    20: { class_type: 'LoadImage', inputs: { image: refA, upload: 'image' } },
    24: { class_type: 'LoadImage', inputs: { image: refB, upload: 'image' } },
    25: { class_type: 'ImageBatch', inputs: { image1: ['20', 0], image2: ['24', 0] } },
    21: { class_type: 'IPAdapterModelLoader', inputs: { ipadapter_file: IPADAPTER } },
    22: { class_type: 'CLIPVisionLoader', inputs: { clip_name: CLIPVISION } },
    23: { class_type: 'IPAdapterAdvanced', inputs: { model: ['4', 0], ipadapter: ['21', 0], image: ['25', 0], weight: ipa.weight, weight_type: ipa.type, combine_embeds: 'concat', start_at: ipa.start, end_at: ipa.end, embeds_scaling: 'K+V', clip_vision: ['22', 0] } },
    3: { class_type: 'KSampler', inputs: { seed, steps: 28, cfg: 6, sampler_name: 'euler_ancestral', scheduler: 'normal', denoise: 1, model: ['23', 0], positive: ['6', 0], negative: ['7', 0], latent_image: ['5', 0] } },
    8: { class_type: 'VAEDecode', inputs: { samples: ['3', 0], vae: ['4', 2] } },
    9: { class_type: 'SaveImage', inputs: { filename_prefix: prefix, images: ['8', 0] } },
  };
  if (cn != null) {
    g[30] = { class_type: 'ControlNetLoader', inputs: { control_net_name: CONTROLNET } };
    g[31] = { class_type: 'LoadImage', inputs: { image: skel, upload: 'image' } };
    g[32] = { class_type: 'ControlNetApplyAdvanced', inputs: { positive: ['6', 0], negative: ['7', 0], control_net: ['30', 0], image: ['31', 0], strength: cn, start_percent: 0, end_percent: 1, vae: ['4', 2] } };
    g[3].inputs.positive = ['32', 0];
    g[3].inputs.negative = ['32', 1];
  }
  return g;
}

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
async function queueState() {
  try {
    const q = await (await fetch(`${COMFY}/queue`)).json();
    return { pending: q.queue_pending?.length || 0, running: q.queue_running?.length || 0 };
  } catch { return { pending: 99, running: 99 }; }
}

async function main() {
  const [sub, arm] = process.argv.slice(2);
  if (!SUBJECTS[sub] || !ARMS[arm]) throw new Error(`usage: render.mjs <${Object.keys(SUBJECTS).join('|')}> <${Object.keys(ARMS).join('|')}>`);
  const S = SUBJECTS[sub];
  const want = flag('n', '1,2,3,4').split(',').map(Number);
  const ipa = { weight: Number(flag('ipa', 0.28)), type: 'ease in', start: 0.2, end: Number(flag('ipaEnd', 0.6)) };
  const skelName = flag('skel', S.skel);
  const tag = flag('tag', '');
  const variant = { pose: flag('pose', null), idDrop: flag('idDrop', ''), idAdd: flag('idAdd', ''), emph: flag('emph', ''), negAdd: flag('negAdd', '') };
  const [W, H] = (flag('size', (S.size || [832, 1216]).join('x'))).split('x').map(Number);
  const seed0 = Number(flag('seed0', S.seed0));
  const cnList = ARMS[arm].cn ? (flag('cn', null) ? flag('cn').split(',').map(Number) : ARMS[arm].cn) : null;
  const folder = `${arm}${tag}`;
  const dir = join(OUT, sub, folder);
  const positive = promptFor(sub, variant);
  const negative = negativeFor(sub, variant);
  const jobs = want.filter((n) => !existsSync(join(dir, `cand-${n}.json`)))
    .map((n) => ({ n, seed: seed0 + n, cn: cnList ? cnList[(n - 1) % cnList.length] : null }));
  if (process.argv.includes('--dry')) {
    console.log(`+ ${positive}\n- ${negative}\n`, jobs);
    return;
  }
  mkdirSync(dir, { recursive: true });
  const refA = stageImage(`${SCRATCH}/refs/${sub}-square.png`);
  const refB = stageImage(`${SCRATCH}/refs/${sub}-head.png`);
  const skel = ARMS[arm].cn ? stageImage(`${SCRATCH}/skeletons/${skelName}.png`) : null;
  const inflight = [];
  let stop = false;
  const tStart = Date.now();
  while ((jobs.length || inflight.length) && !stop) {
    while (jobs.length && inflight.length < MAX_PENDING) {
      const q = await queueState();
      if (q.pending >= MAX_PENDING) break;
      const j = jobs.shift();
      const g = graph({ positive, negative, seed: j.seed, cn: j.cn, skel, refA, refB, ipa, width: W, height: H, prefix: `pyrefly/persp-2026-09-27/${sub}-${folder}` });
      const res = await fetch(`${COMFY}/prompt`, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ prompt: g, client_id: 'pyrefly-persp-art' }) });
      const body = await res.json();
      if (!res.ok || body.error) throw new Error(`ComfyUI rejected: ${JSON.stringify(body).slice(0, 1500)}`);
      inflight.push({ ...j, id: body.prompt_id, t0: Date.now() });
    }
    await sleep(2000);
    for (const j of [...inflight]) {
      const e = (await (await fetch(`${COMFY}/history/${j.id}`)).json())[j.id];
      if (!e) continue;
      if (e.status?.status_str === 'error') { console.error(`[persp-art] ${sub}/${folder}#${j.n} FAILED ${JSON.stringify(e.status.messages).slice(0, 800)}`); inflight.splice(inflight.indexOf(j), 1); continue; }
      const img = Object.values(e.outputs || {}).flatMap((o) => o.images || [])[0];
      if (!img) continue;
      inflight.splice(inflight.indexOf(j), 1);
      const qs = new URLSearchParams({ filename: img.filename, subfolder: img.subfolder || '', type: img.type || 'output' });
      const raw = join(dir, `cand-${j.n}.raw.png`);
      const cut = join(dir, `cand-${j.n}.png`);
      writeFileSync(raw, Buffer.from(await (await fetch(`${COMFY}/view?${qs}`)).arrayBuffer()));
      const maxRgb = maxRgbOfPng(readFileSync(raw));
      const black = maxRgb != null && isBlackFrame(maxRgb);
      if (black) {
        writeFileSync(join(OUT, 'STOP-BLACK.txt'), `${new Date().toISOString()} ${sub}/${folder}#${j.n} all-black frame (maxRgb ${maxRgb}); run stopped, no re-roll\n`, { flag: 'a' });
        console.error(`[persp-art] BLACK FRAME ${sub}/${folder}#${j.n}: stopping`);
        stop = true;
      }
      let cutMeta = null; let guard = { ok: false, reasons: ['black frame'] };
      if (!black) {
        cutMeta = cutout(raw, cut);
        guard = await checkCutoutFile(cut, { sourceWidth: W, sourceHeight: H, composition: S.framing ? 'boss' : 'full' });
      }
      const side = {
        round: 'perspectives-2026-09-27', game: S.game, subject: sub, view: 'rear34', arm: folder, candidate: j.n, seed: j.seed,
        status: 'CANDIDATE', requestedFacing: S.framing ? 'away-left (rear view of a boss, faces into the frame toward screen-left)' : 'away-right (rear three-quarter, faces into the frame toward screen-right)', chiral: S.chiral,
        method: ARMS[arm].cn ? 'B: txt2img + rear phrasing + Method F reference + OpenPose rear skeleton' : 'A: txt2img + rear phrasing + Method F reference',
        controlnet: j.cn != null ? { file: CONTROLNET, strength: j.cn, start: 0, end: 1, skeleton: `${skelName}.png (tools/zz-persp-art/skeletons.py)` } : null,
        ipadapter: { file: IPADAPTER, ...ipa, scaling: 'K+V', combine: 'concat', images: [`${sub}-square.png`, `${sub}-head.png`] },
        variant, positive, negative, model: CKPT, steps: 28, cfg: 6, sampler: 'euler_ancestral', scheduler: 'normal', canvas: { width: W, height: H },
        maxRgb, black, cutout: cutMeta, guard: { ok: guard.ok, reasons: guard.reasons },
        seconds: (Date.now() - j.t0) / 1000, generatedAt: new Date().toISOString(),
      };
      if (cutMeta) Object.assign(side, { width: cutMeta.width, height: cutMeta.height, baselineY: cutMeta.baselineY });
      writeFileSync(join(dir, `cand-${j.n}.json`), JSON.stringify(side, null, 1));
      console.log(`[persp-art] ${sub}/${folder}#${j.n} seed ${j.seed}${j.cn != null ? ` cn ${j.cn}` : ''}: ${cutMeta ? `${cutMeta.width}x${cutMeta.height}` : '-'} guard ${guard.ok ? 'ok' : 'REJECT ' + guard.reasons.join('; ')}${black ? ' BLACK' : ''} ${side.seconds.toFixed(0)} s`);
    }
  }
  console.log(`[persp-art] ${sub}/${folder}: wall ${((Date.now() - tStart) / 1000).toFixed(0)} s`);
  if (stop) process.exit(3);
}

const isMain = process.argv[1] && resolve(process.argv[1]) === fileURLToPath(import.meta.url);
if (isMain) await main();
