def P(o, length, deg):
    """Point `length` px from `o` at screen angle `deg` (0 right, 90 down, 180 left, 270 up)."""
    r = math.radians(deg)
    return (o[0] + length * math.cos(r), o[1] + length * math.sin(r))


def rear_body(hip, *, s=1.0, sh=46, hp=34, rarm=(88, 80), larm=(96, 94), rleg=(88, 90), lleg=(94, 92),
              head_dx=10, ear_dx=22, ear_dy=-66, torso_lean=0):
    """Forward kinematics for a rear three-quarter figure facing into the frame toward screen-right.
    rarm/larm/rleg/lleg = (upper, lower) screen angles. The RIGHT side is screen-right (near side)."""
    T, UA, FA, TH, SH = 250 * s, 150 * s, 135 * s, 265 * s, 255 * s
    k = {}
    neck = P(hip, T, 270 + torso_lean)
    k[1] = neck
    k[2] = (neck[0] + sh * s, neck[1])        # right shoulder: screen-right (seen from behind)
    k[5] = (neck[0] - sh * s, neck[1])        # left shoulder: screen-left
    k[8] = (hip[0] + hp * s, hip[1])          # right hip
    k[11] = (hip[0] - hp * s, hip[1])         # left hip
    k[3] = P(k[2], UA, rarm[0]); k[4] = P(k[3], FA, rarm[1])
    k[6] = P(k[5], UA, larm[0]); k[7] = P(k[6], FA, larm[1])
    k[9] = P(k[8], TH, rleg[0]); k[10] = P(k[9], SH, rleg[1])
    k[12] = P(k[11], TH, lleg[0]); k[13] = P(k[12], SH, lleg[1])
    head = (neck[0] + head_dx * s, neck[1] + ear_dy * s)
    k[16] = (head[0] + ear_dx * s, head[1])   # right ear only (near side); no nose, no eyes, no left ear
    return k


def ground_scale(size, k, f, gy=1170):
    cx = size[0] / 2
    return {i: (cx + (x - cx) * f, gy + (y - gy) * f) for i, (x, y) in k.items()}

def poses():
    out = {n: (size, ground_scale(size, k, SCALE)) for n, (size, k) in poses_raw().items()}
    out.update(poses_v2())
    return out


def draw(size, pts, stick=8, radius=7):
    W, H = size
    over = Image.new('RGB', (W, H), (0, 0, 0))
    d = ImageDraw.Draw(over)
    for li, (a, b) in enumerate(LIMBS):
        if a not in pts or b not in pts:
            continue
        (x1, y1), (x2, y2) = pts[a], pts[b]
        length = math.hypot(x2 - x1, y2 - y1)
        ang = math.atan2(y2 - y1, x2 - x1)
        mx, my = (x1 + x2) / 2, (y1 + y2) / 2
        poly = []
        for i in range(36):
            th = 2 * math.pi * i / 36
            ex, ey = length / 2 * math.cos(th), stick * math.sin(th)
            poly.append((mx + ex * math.cos(ang) - ey * math.sin(ang), my + ex * math.sin(ang) + ey * math.cos(ang)))
        d.polygon(poly, fill=COLORS[li])
    img = Image.blend(Image.new('RGB', (W, H)), over, 0.6)
    d2 = ImageDraw.Draw(img)
    for kk, (x, y) in pts.items():
        d2.ellipse((x - radius, y - radius, x + radius, y + radius), fill=COLORS[kk])
    return img


def main(only=None):
    OUT.mkdir(parents=True, exist_ok=True)
    tiles = []
    for name, (size, pts) in poses().items():
        if only and name not in only:
            continue
        im = draw(size, pts)
        im.save(OUT / f'{name}.png')
        oob = [k for k, (x, y) in pts.items() if not (0 <= x < size[0] and 0 <= y < size[1])]
        print(name, size, 'out of frame:', oob or 'none')
        tiles.append((name, im))
    h = 480
    row = [t.resize((int(t.width * h / t.height), h)) for _, t in tiles]
    sheet = Image.new('RGB', (sum(r.width for r in row) + 6 * len(row), h), (60, 60, 60))
    x = 0
    for r in row:
        sheet.paste(r, (x, 0)); x += r.width + 6
    sheet.save(OUT / 'overview.jpg', quality=85)


if __name__ == '__main__':
    main(sys.argv[1:] or None)
