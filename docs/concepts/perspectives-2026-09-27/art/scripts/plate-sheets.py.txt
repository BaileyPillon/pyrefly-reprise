def font(size):
    for f in ("arialbd.ttf", "arial.ttf"):
        try:
            return ImageFont.truetype(f, size)
        except Exception:
            pass
    return ImageFont.load_default()


def tile(path, label, sub, pick=False):
    im = Image.open(path).convert("RGB").resize((TW, TH), Image.LANCZOS)
    t = Image.new("RGB", (TW + 12, TH + 12 + 44), (34, 34, 38))
    t.paste(im, (6, 50))
    d = ImageDraw.Draw(t)
    if pick:
        d.rectangle([0, 0, t.width - 1, t.height - 1], outline=GOLD, width=5)
    d.text((8, 4), label, fill=GOLD if pick else (235, 235, 235), font=font(17))
    d.text((8, 25), sub, fill=GOLD if pick else (175, 175, 175), font=font(14))
    return t


def main():
    for nn, name, pick, verdicts, why in SHEETS:
        tiles = [tile(PLATES / f"{name}.png", "TODAY'S PLATE (shipped)", "the view the camera sees today")]
        for arm in sorted(p for p in (CAND / name).iterdir() if p.is_dir()):
            for cj in sorted(arm.glob("cand-*.json"), key=lambda p: int(p.stem.split("-")[1])):
                m = json.loads(cj.read_text(encoding="utf-8"))
                key = f"{arm.name}/{m['candidate']}"
                is_pick = f"{arm.name}/{cj.stem}" == pick
                tiles.append(tile(cj.with_suffix(".png"), f"{'PICK  ' if is_pick else ''}{arm.name} #{m['candidate']}  s{m['seed']}",
                                  "installed plate-reverse.png" if is_pick else verdicts.get(key, ""), is_pick))
                graded = cj.with_name(f"{cj.stem}.graded.png")
                if graded.exists():
                    tiles.append(tile(graded, f"{arm.name} #{m['candidate']} graded", verdicts.get(f"{key}g", "")))
        rows = [tiles[i:i + PER_ROW] for i in range(0, len(tiles), PER_ROW)]
        W = PER_ROW * (tiles[0].width + 10) + 10
        H = 56 + len(rows) * (tiles[0].height + 10) + 80
        sheet = Image.new("RGB", (W, H), (22, 22, 26))
        d = ImageDraw.Draw(sheet)
        d.text((12, 14), f"{name}: reverse-angle plate candidates, every render (perspectives round 2026-09-27, art agent)",
               fill=(240, 240, 240), font=font(24))
        y = 56
        for r in rows:
            x = 10
            for t in r:
                sheet.paste(t, (x, y))
                x += t.width + 10
            y += tiles[0].height + 10
        for i, line in enumerate(textwrap.wrap(why, int(W / 10.5))[:3]):
            d.text((12, y + 6 + 24 * i), line, fill=GOLD, font=font(18))
        out = OUT / f"sheet-{nn}-plate-{name}.jpg"
        sheet.save(out, quality=88)
        print(out.name, sheet.size, len(tiles) - 1, "tiles")


if __name__ == "__main__":
    main()
