"""Quick look strip: images side by side at one height on a mid-grey ground, labelled.
    python look.py <out.jpg> <height> <label=path> [<label=path> ...]
"""
import sys
from PIL import Image, ImageDraw, ImageFont
out, h = sys.argv[1], int(sys.argv[2])
items = [a.split('=', 1) for a in sys.argv[3:]]
tiles = []
for label, p in items:
    im = Image.open(p).convert('RGBA')
    bb = im.getchannel('A').getbbox() if im.mode == 'RGBA' else None
    s = h / im.height
    im = im.resize((max(1, int(im.width * s)), h), Image.LANCZOS)
    bg = Image.new('RGBA', im.size, (118, 118, 118, 255))
    bg.alpha_composite(im)
    tiles.append((label, bg.convert('RGB')))
W = sum(t.width for _, t in tiles) + 10 * (len(tiles) + 1)
sheet = Image.new('RGB', (W, h + 40), (40, 40, 40))
d = ImageDraw.Draw(sheet)
try:
    f = ImageFont.truetype('arial.ttf', 22)
except Exception:
    f = ImageFont.load_default()
x = 10
for label, t in tiles:
    sheet.paste(t, (x, 36))
    d.text((x + 4, 6), label, fill=(255, 255, 255), font=f)
    x += t.width + 10
sheet.save(out, quality=90)
print(sheet.size)
