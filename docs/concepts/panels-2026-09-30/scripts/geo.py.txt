# HUD geometry measured on the real-engine frames (1920x1080), used to cut HUD pieces with polygon masks
# and to paint the HUD out of today's frames. Measured by edge scans + overlays (see z-geo-*.png in scratch).
import math
T12 = math.tan(math.radians(12))


def slab(xbl, ytop, w, h, slant=1):
    o = h * T12
    if slant >= 0:
        return [(xbl + o, ytop), (xbl + w + o, ytop), (xbl + w, ytop + h), (xbl, ytop + h)]
    return [(xbl, ytop), (xbl + w, ytop), (xbl + w + o, ytop + h), (xbl + o, ytop + h)]


def grow(pts, d):
    """Crude polygon offset: push every vertex away from the centroid by d px (enough for masks)."""
    cx = sum(p[0] for p in pts) / len(pts); cy = sum(p[1] for p in pts) / len(pts)
    out = []
    for x, y in pts:
        vx, vy = x - cx, y - cy
        n = math.hypot(vx, vy) or 1
        out.append((x + vx / n * d * 1.4, y + vy / n * d * 1.4))
    return out
